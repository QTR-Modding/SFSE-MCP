cmake_minimum_required(VERSION 3.25)
project(SFSEMCPInstallConsumer LANGUAGES CXX)
find_package(SFSE-MCP CONFIG REQUIRED)
if(NOT SFSEMCP_SIGNING_INCLUDE_DIR STREQUAL EXPECTED_SIGNING_INCLUDE_DIR)
    message(FATAL_ERROR "Consumer found signing headers outside the expected installation")
endif()
add_executable(install_consumer ../header_compile.cpp ../header_odr.cpp)
target_link_libraries(install_consumer PRIVATE SFSE-MCP::SFSE-MCP)
if(MSVC)
    target_compile_options(install_consumer PRIVATE /W4 /WX /permissive- /EHsc)
endif()
enable_testing()
add_test(NAME installed_sdk COMMAND install_consumer)
